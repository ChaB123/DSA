Solution #1 : Iterative :
Time Complexity : O(n), Space Complexity : O(n) :

class Solution {
    public void reorderList(ListNode head) {
        if(head == null || head.next == null || head.next.next == null){
            return;
        }

        int size = 0;
        ListNode tmp = head;
        while(tmp != null){
            size += 1;
            tmp = tmp.next;
        }

        ListNode[] storeNodes = new ListNode[size];
        tmp = head;
        size = 0;
        while(tmp != null){
            storeNodes[size] = tmp;
            size += 1;
            tmp = tmp.next;
        }

        int i = 0;
        for(i=0;i<size/2;i++){
            storeNodes[i].next = storeNodes[size-i-1];
            storeNodes[size-i-1].next = storeNodes[i+1];
        }
        storeNodes[i].next = null;
    }
}

class Solution {
    public void reorderList(ListNode head) {
        if(head == null || head.next == null || head.next.next == null){
            return;
        }

        int size = 0;
        ListNode tmp = head;
        while(tmp != null){
            size += 1;
            tmp = tmp.next;
        }

        int halfSize = (size/2)-1;
        tmp = head;

        for(int i=0;i<halfSize;i++){
            tmp = tmp.next;
        }
        ListNode headOfReversedSecondHalf = reverse(tmp.next);
        tmp.next = null;

        reorderTwoHalfs(head, headOfReversedSecondHalf);
    }
    public ListNode reverse(ListNode head){
        if(head == null || head.next == null){
            return head;
        }

        ListNode previous = null;
        ListNode current = head;
        ListNode nextElement = head.next;

        while(nextElement != null){
            current.next = previous;
            previous = current;
            current = nextElement;
            nextElement = nextElement.next;
        }

        current.next = previous;
        return current;
    }
    public void reorderTwoHalfs(ListNode start, ListNode midStart){
        while(start.next != null && midStart != null){
            ListNode startNext = start.next;
            start.next = midStart;

            ListNode midNext = midStart.next;
            midStart.next = startNext;

            start = startNext;
            midStart = midNext;
        }

        start.next = midStart;
    }
}

class Solution {
    public void reorderList(ListNode head) {
        if(head == null || head.next == null || head.next.next == null){
            return;
        }

        ListNode slow = head;
        ListNode fast = head;

        while(fast.next != null && fast.next.next != null){
            slow = slow.next;
            fast = fast.next.next;
        }

        ListNode headOfReversedSecondHalf = reverse(slow.next);
        slow.next = null;

        ListNode start = head;
        while(headOfReversedSecondHalf != null){
            ListNode startNext = start.next;
            ListNode midNext = headOfReversedSecondHalf.next;

            start.next = headOfReversedSecondHalf;
            headOfReversedSecondHalf.next = startNext;

            start = startNext;
            headOfReversedSecondHalf = midNext;
        }
    }
    public ListNode reverse(ListNode head){
        if(head == null || head.next == null){
            return head;
        }

        ListNode previous = null;
        ListNode current = head;
        ListNode nextElement = head.next;

        while(nextElement != null){
            current.next = previous;
            previous = current;
            current = nextElement;
            nextElement = nextElement.next;
        }

        current.next = previous;
        return current;
    }
}
