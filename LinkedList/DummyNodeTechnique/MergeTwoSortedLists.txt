Solution #1 : Iterative :
Time Complexity : O(m+n), Space Complexity : O(1) :

class Solution {
    public ListNode mergeTwoLists(ListNode list1, ListNode list2) {
        if(list1 == null){
            return list2;
        }

        if(list2 == null){
            return list1;
        }

        ListNode toReturn = null;
        ListNode dummy = null;

        if(list1.val < list2.val){
            toReturn = list1;
            dummy = list1;
            list1 = list1.next;
        }else{
            toReturn = list2;
            dummy = list2;
            list2 = list2.next;
        }

        while(list1 != null && list2 != null){
            if(list1.val < list2.val){
                dummy.next = list1;
                list1 = list1.next;
            }else{
                dummy.next = list2;
                list2 = list2.next;
            }

            dummy = dummy.next;
        }

        if(list1 != null){
            dummy.next = list1;
        }

        if(list2 != null){
            dummy.next = list2;
        }

        return toReturn;
    }
}
