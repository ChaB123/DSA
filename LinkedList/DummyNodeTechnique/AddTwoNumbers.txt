LeetCode : https://leetcode.com/problems/add-two-numbers/description/

Solution #1 : Iterative :
Time Complexity : O(m+n), Space Complexity : O(m+n) :

class Solution {
    public ListNode addTwoNumbers(ListNode l1, ListNode l2) {
        ListNode res = new ListNode();
        ListNode dummy = new ListNode();
        dummy = res;
        int remainder = 0;

        while(l1 != null && l2 != null){
            int sum = remainder+l1.val+l2.val;
            if(sum > 9){
                remainder = sum/10;
                sum = sum%10;
            }else{
                remainder = 0;
            }

            ListNode test = new ListNode(sum, null);
            res.next = test;
            res = res.next;

            l1 = l1.next;
            l2 = l2.next;
        }

        while(l1 != null){
            int sum = remainder+l1.val;
            if(sum > 9){
                remainder = sum/10;
                sum = sum%10;
            }else{
                remainder = 0;
            }

            ListNode test = new ListNode(sum, null);
            res.next = test;
            res = res.next;

            l1 = l1.next;
        }

        while(l2 != null){
            int sum = remainder+l2.val;
            if(sum > 9){
                remainder = sum/10;
                sum = sum%10;
            }else{
                remainder = 0;
            }

            ListNode test = new ListNode(sum, null);
            res.next = test;
            res = res.next;

            l2 = l2.next;
        }

        if(remainder > 0){
            ListNode test = new ListNode(remainder, null);
            res.next = test;
        }

        return dummy.next;
    }
}
