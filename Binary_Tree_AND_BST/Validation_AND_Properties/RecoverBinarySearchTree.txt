Solution #1 : 
Time Complexity : O(nlogn), Space Complexity : O(n) :

class Solution {
    List<Integer> tmp = new ArrayList<>();
    int index = 0;
    public void recoverTree(TreeNode root) {
        inOrder(root);
        Collections.sort(tmp); 
        inOrderFix(root);
    }
    public void inOrder(TreeNode root){
        if(root == null){
            return;
        }

        inOrder(root.left);
        tmp.add(root.val);
        inOrder(root.right);
    }
    public void inOrderFix(TreeNode root){
        if(root == null){
            return;
        }

        inOrderFix(root.left);
        root.val = tmp.get(index);
        index += 1;
        inOrderFix(root.right);
    }
}

Solution #2 : 
Time Complexity : O(n), Space Complexity : O(h) :

class Solution {
    TreeNode prev = null;
    TreeNode first = null;
    TreeNode second = null;
    public void recoverTree(TreeNode root) {
        inOrder(root);
        int tmp = first.val;
        first.val = second.val;
        second.val = tmp;
    }
    public void inOrder(TreeNode root){
        if(root == null){
            return;
        }

        inOrder(root.left);
        if(prev != null && prev.val >= root.val){
            if(first == null){
                first = prev;
            }
            second = root;
        }

        prev = root;
        inOrder(root.right);
    }
}
