Solution #1 :
Time Complexity : O(n), Space Complexity : O(h) :

class Solution {
    int sumOfTilt = 0;
    public int findTilt(TreeNode root) {
        findSubTreeSum(root);
        return sumOfTilt;
    }
    public int findSubTreeSum(TreeNode root){
        if(root == null){
            return 0;
        }

        int sumOfLeftSubTree = findSubTreeSum(root.left);
        int sumOfRightSubTree = findSubTreeSum(root.right);
        sumOfTilt += Math.abs(sumOfLeftSubTree - sumOfRightSubTree);

        return sumOfLeftSubTree + sumOfRightSubTree + root.val;
    }
}
