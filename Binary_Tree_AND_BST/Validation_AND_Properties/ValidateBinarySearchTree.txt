LeetCode : https://leetcode.com/problems/validate-binary-search-tree/description/

Solution #1 :
Time Complexity : O(n), Space Complexity : O(h) :

class Solution {
    int rightVal = Integer.MIN_VALUE;
    boolean isThisRightMostValue = true;
    public boolean isValidBST(TreeNode root) {
        if(root == null){
            return true;
        }

        boolean rightAns = isValidBST(root.right);

        if(!rightAns){
            return false;
        }

        if(rightVal == Integer.MIN_VALUE && isThisRightMostValue){
            rightVal = root.val;
            isThisRightMostValue = false;
        }else{
            if(rightVal <= root.val){
                return false;
            }else{
                rightVal = root.val;
            }
        }

        return isValidBST(root.left);
    }
}

Solution #2 : More cleaner code :
Time Complexity : O(n), Space Complexity : O(h) :

class Solution {
    Integer rightVal = null;
    public boolean isValidBST(TreeNode root) {
        if(root == null){
            return true;
        }

        if(!isValidBST(root.right)){
            return false;
        }

        if(rightVal != null && rightVal <= root.val){
            return false;
        }

        rightVal = root.val;

        return isValidBST(root.left);
    }
}

Solution #3 : Morris traversal
Time Complexity : O(n), Space Complexity : O(1)

class Solution {
    public boolean isValidBST(TreeNode root) {
        Integer rightVal = null;
        TreeNode current = root;

        while(current != null){
            if(current.left == null){
                if(rightVal != null && rightVal >= current.val){
                    return false;
                }

                rightVal = current.val;
                current = current.right;
            }else{
                //Find inorder predecessor of current
                TreeNode prev = current.left;
                while(prev.right != null && prev.right != current){
                    prev = prev.right;
                }

                //Make current the right child of its inorder predecessor
                if(prev.right == null){
                    prev.right = current;
                    current = current.left;
                }else{
                    //Revert the changes made in the tree structure above
                    prev.right = null;

                    if(rightVal != null && rightVal >= current.val){
                        return false;
                    }
                    rightVal = current.val;
                    current = current.right;
                }
            }
        }
        return true;
    }
}
