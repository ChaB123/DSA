LeetCode : https://leetcode.com/problems/path-sum-ii/description/

Solution #1 :
Time Complexity : O(n), Space Complexity : O(h*k) (h = height of tree, k = maximum number of valid paths) :

class Solution {
    List<List<Integer>> ans = new ArrayList<>();
    public List<List<Integer>> pathSum(TreeNode root, int targetSum) {
        findAllTargetSumPath(root, targetSum, new ArrayList<>());
        return ans;
    }

    public void findAllTargetSumPath(TreeNode root, int targetSum, List<Integer> previousProcessedValues){
        if(root == null){
            return;
        }

        previousProcessedValues.add(root.val);

        if(root.left == null && root.right == null && root.val == targetSum){
            ans.add(new ArrayList<>(previousProcessedValues));
            previousProcessedValues.remove(previousProcessedValues.size()-1);
            return;
        }

        
        findAllTargetSumPath(root.left, targetSum - root.val, previousProcessedValues);
        findAllTargetSumPath(root.right, targetSum - root.val, previousProcessedValues);
        previousProcessedValues.remove(previousProcessedValues.size()-1);
    }
}

Solution #2 : More cleaner code (removing global variable)
Time Complexity : O(n), Space Complexity : O(h*k) :

class Solution {
    
    public List<List<Integer>> pathSum(TreeNode root, int targetSum) {
        List<List<Integer>> ans = new ArrayList<>();
        findAllTargetSumPath(root, targetSum, new ArrayList<>(), ans);
        return ans;
    }

    public void findAllTargetSumPath(TreeNode root, int targetSum, List<Integer> previousProcessedValues, List<List<Integer>> ans){
        if(root == null){
            return;
        }

        previousProcessedValues.add(root.val);

        if(root.left == null && root.right == null && root.val == targetSum){
            ans.add(new ArrayList<>(previousProcessedValues));
            previousProcessedValues.remove(previousProcessedValues.size()-1);
            return;
        }

        
        findAllTargetSumPath(root.left, targetSum - root.val, previousProcessedValues, ans);
        findAllTargetSumPath(root.right, targetSum - root.val, previousProcessedValues, ans);
        previousProcessedValues.remove(previousProcessedValues.size()-1);
    }
}
