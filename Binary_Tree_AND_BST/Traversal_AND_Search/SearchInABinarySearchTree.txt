LeetCode : https://leetcode.com/problems/search-in-a-binary-search-tree/description/

Solution #1 :
Time Complexity : O(h) (h = heigth of BST, Balanced BST: h = log n → O(log n), Skewed BST: h = n → O(n)), Space Complexity : O(h) :

class Solution {
    public TreeNode searchBST(TreeNode root, int val) {
        if(root == null){
            return null;
        }

        if(root.val == val){
            return root;
        }

        if(val > root.val){
            return searchBST(root.right, val);
        }

        return searchBST(root.left, val);
    }
}
