Solution #1 : 
Time Complexity : O(n), Space Complexity : O(n) :

class Solution {
    public TreeNode sortedListToBST(ListNode head) {
        List<Integer> storeAllElement = new ArrayList<>();
        ListNode newHead = head;

        while(newHead != null){
            storeAllElement.add(newHead.val);
            newHead = newHead.next;
        }

        return constructBSTFromArrayList(storeAllElement, 0, storeAllElement.size()-1);
    }
    public TreeNode constructBSTFromArrayList(List<Integer> storeAllElement, int start, int end){
        if(start > end){
            return null;
        }

        int rootNodeIndex = (start+end)/2;

        if((end-start)%2 != 0){
            rootNodeIndex += 1;
        }

        TreeNode currentNode = new TreeNode(storeAllElement.get(rootNodeIndex));
        currentNode.left = constructBSTFromArrayList(storeAllElement, start, rootNodeIndex-1);
        currentNode.right = constructBSTFromArrayList(storeAllElement, rootNodeIndex+1, end);

        return currentNode;
    }
}
