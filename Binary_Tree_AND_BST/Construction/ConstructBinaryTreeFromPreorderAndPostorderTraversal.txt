LeetCode : https://leetcode.com/problems/construct-binary-tree-from-preorder-and-postorder-traversal/description/

Solution #1 : 
Time Complexity : O(n), Space Complexity : O(n) :

class Solution {
    int postOrderIndex = 0;
    public TreeNode constructFromPrePost(int[] preorder, int[] postorder) {
        postOrderIndex = postorder.length-1;

        HashMap<Integer, Integer> elementsIndex = new HashMap<>();
        for(int i=0;i<preorder.length;i++){
            elementsIndex.put(preorder[i], i);
        }

        return contructBinaryTreeWithPrePost(preorder, postorder, 0, preorder.length-1, elementsIndex);
    }

    public TreeNode contructBinaryTreeWithPrePost(int[] preorder, int[] postorder, int start, int end, HashMap<Integer, Integer> elementsIndex){
        if(start > end){
            return null;
        }

        if(start == end){
            TreeNode currentRoot = new TreeNode(postorder[postOrderIndex], null, null);
            postOrderIndex -= 1;
            return currentRoot;
        }

        TreeNode currentRoot = new TreeNode(postorder[postOrderIndex]);
        postOrderIndex -= 1;

        int nextNodeIndex = elementsIndex.get(postorder[postOrderIndex]);

        currentRoot.right = contructBinaryTreeWithPrePost(preorder, postorder, nextNodeIndex, end, elementsIndex);
        currentRoot.left = contructBinaryTreeWithPrePost(preorder, postorder, elementsIndex.get(currentRoot.val)+1, nextNodeIndex-1, elementsIndex);

        return currentRoot;
    }
}
