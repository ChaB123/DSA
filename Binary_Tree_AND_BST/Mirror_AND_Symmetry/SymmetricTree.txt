Solution #1 :
Time Complexity : O(n), Space Complexity : O(n) :

class Solution {
    public boolean isSymmetric(TreeNode root) {
        if(root == null){
            return true;
        }
        
        return checkSymmetry(root.left, root.right);
    }

    public boolean checkSymmetry(TreeNode leftElement, TreeNode rightElement){
        if(leftElement == null || rightElement == null){
            return leftElement == rightElement;
        }

        if(leftElement.val != rightElement.val){
            return false;
        }

        if(!checkSymmetry(leftElement.left, rightElement.right)){
            return false;
        }

        if(!checkSymmetry(leftElement.right, rightElement.left)){
            return false;
        }

        return true;
    }
}
