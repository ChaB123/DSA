Solution #1 :
Time Complexity : O(nlogK), Space Complexity : O(K) :

class Solution {
    public ListNode mergeKLists(ListNode[] lists) {
        ListNode root = null;
        ListNode previousNode = null;
        PriorityQueue<ListNode> store = new PriorityQueue<>((a,b) -> {
            return Integer.compare(a.val, b.val);
        });

        for(int i=0;i<lists.length;i++){
            if(lists[i] != null){
                store.add(lists[i]);
            }
        }

        while(!store.isEmpty()){
            ListNode currentNode = store.poll();

            if(root == null){
                root = currentNode;
                previousNode = root;
            }else{
                previousNode.next = currentNode;
                previousNode = previousNode.next;
            }

            if(currentNode.next != null){
                store.add(currentNode.next);
            }
        }

        return root;
    }
}
