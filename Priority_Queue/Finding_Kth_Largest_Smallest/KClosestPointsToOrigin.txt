LeetCode : https://leetcode.com/problems/k-closest-points-to-origin/description/

Solution #1 : 
Time Complexity : O(nlogK), Space Complexity : O(K) :

class Solution {
    public int[][] kClosest(int[][] points, int k) {
        PriorityQueue<int[]> maxHeap = new PriorityQueue<>(
            (a, b) -> Integer.compare(b[0], a[0])
        );


        for(int i=0;i<points.length;i++){
            int distance = computeDistance(points[i][0], points[i][1]);
            maxHeap.add(new int[]{distance, points[i][0], points[i][1]});

            while(maxHeap.size() > k){
                maxHeap.poll();
            }
        }

        int[][] result = new int[k][2];
        int index = 0;

        while(!maxHeap.isEmpty()){
            int[] currentVal = maxHeap.poll();
            result[index][0] = currentVal[1];
            result[index][1] = currentVal[2];

            index += 1;
        }

        return result;
    }
    public int computeDistance(int x, int y){
        int distance = x*x + y*y;

        return distance;
    }
}
