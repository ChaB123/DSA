LeetCode : https://leetcode.com/problems/sort-array-by-increasing-frequency/description/

Solution #1 : 
Time Complexity : O(nlogn), Space Complexity : O(n) :

class Solution {
    public int[] frequencySort(int[] nums) {
        HashMap<Integer, Integer> freq = new HashMap<>();
        for(int i=0;i<nums.length;i++){
            if(freq.containsKey(nums[i])){
                freq.put(nums[i], freq.get(nums[i])+1);
            }else{
                freq.put(nums[i], 1);
            }
        }

        PriorityQueue<int[]> store = new PriorityQueue<>((a, b) -> {
            int freqCompare = Integer.compare(
                a[1], b[1]
            );

            if (freqCompare != 0) {
                return freqCompare;
            }

            // Same frequency:
            // the larger key should come first in the min-heap
            return Integer.compare(b[0], a[0]);
        });

        for(int key : freq.keySet()){
            int[] elementAndFrequency = new int[2];
            elementAndFrequency[0] = key;
            elementAndFrequency[1] = freq.get(key);

            store.add(elementAndFrequency);
        }

        int[] ans = new int[nums.length];
        int index = 0;

        while(!store.isEmpty()){
            int[] currentElement = store.poll();
            for(int i=0;i<currentElement[1];i++){
                ans[index] = currentElement[0];
                index += 1;
            }
        }

        return ans;
    }
}
