Solution #1 : 
Time Complexity : O(nlogK), Space Complexity : O(n) :

class Solution {
    public List<String> topKFrequent(String[] words, int k) {
        HashMap<String, Integer> freq = new HashMap<>();

        for(int i=0;i<words.length;i++){
            if(freq.containsKey(words[i])){
                freq.put(words[i], freq.get(words[i])+1);
            }else{
                freq.put(words[i], 1);
            }
        }

        PriorityQueue<String[]> store = new PriorityQueue<>((a, b) -> {
            int freqCompare = Integer.compare(
                Integer.parseInt(a[1]),
                Integer.parseInt(b[1])
            );

            if (freqCompare != 0) {
                return freqCompare;
            }

            // Same frequency:
            // lexicographically larger word should come first in the min-heap
            return b[0].compareTo(a[0]);
        });

        for(String key : freq.keySet()){
            String[] toAdd = new String[2];
            toAdd[0] = key;
            toAdd[1] = freq.get(key)+"";

            store.add(toAdd);

            while(store.size() > k){
                store.poll();
            }
        }

        List<String> ans = new ArrayList<>();

        while(!store.isEmpty()){
            String[] tmp = store.poll();
            ans.add(tmp[0]);
        }

        //reverse the ans because strings are store with lowest -> heighest frequency
        Collections.reverse(ans);

        return ans;
    }
}
