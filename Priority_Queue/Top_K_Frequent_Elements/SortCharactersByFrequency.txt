Solution #1 : 
Time Complexity : O(nlogn), Space Complexity : O(n) :

class Solution {
    public String frequencySort(String s) {
        HashMap<Character, Integer> freq = new HashMap<>();

        for(int i=0;i<s.length();i++){
            if(freq.containsKey(s.charAt(i))){
                freq.put(s.charAt(i), freq.get(s.charAt(i))+1);
            }else{
                freq.put(s.charAt(i), 1);
            }
        }

        PriorityQueue<char[]> store = new PriorityQueue<>((a,b) -> Integer.compare(b[1]-'0', a[1]-'0'));

        for(char key : freq.keySet()){
            char[] toAdd = new char[2];
            toAdd[0] = key;
            toAdd[1] = (char) (freq.get(key)+'0');

            store.add(toAdd);
        }

        StringBuilder sb = new StringBuilder();
        while(!store.isEmpty()){
            char[] keyValue = store.poll();

            int numberOfTimes = keyValue[1] - '0';
            char key = keyValue[0];
            
            for(int i=0;i<numberOfTimes;i++){
                sb.append(key);
            }
        }

        return sb.toString();
    }
}

Solution #2 :
Time Complexity : O(n), Space Complexity : O(n) :

class Solution {
    public String frequencySort(String s) {
        HashMap<Character, Integer> freq = new HashMap<>();

        for(int i=0;i<s.length();i++){
            if(freq.containsKey(s.charAt(i))){
                freq.put(s.charAt(i), freq.get(s.charAt(i))+1);
            }else{
                freq.put(s.charAt(i), 1);
            }
        }

        List<Character>[] store = new List[s.length()+1];

        for(char key : freq.keySet()){
            int frequencyOfKey = freq.get(key);

            if(store[frequencyOfKey] == null){
                store[frequencyOfKey] = new ArrayList<>();
            }

            store[frequencyOfKey].add(key);
        }

        StringBuilder sb = new StringBuilder();
        for(int i=store.length-1;i>=0;i--){
            if(store[i] == null){
                continue;
            }

            List<Character> numberOfCharsToAdd = store[i];
            for(int j=0;j<numberOfCharsToAdd.size();j++){
                for(int z=0;z<i;z++){
                    sb.append(numberOfCharsToAdd.get(j));
                }
            }
        }

        return sb.toString();
    }
}
