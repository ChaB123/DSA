LeetCode : https://leetcode.com/problems/largest-rectangle-in-histogram/description/

Solution #1 : Iterative :
Time Complexity : O(n), Space Complexity : O(n)

class Solution {
    public int largestRectangleArea(int[] heights) {
        int n = heights.length;

        int[] leftExtend = new int[heights.length];
        int[] rightExtend = new int[heights.length];

        Stack<Integer> store = new Stack<>();

        //calculate how much left does this height extend to
        for(int i=0;i<n;i++){
            while(!store.isEmpty() && heights[store.peek()] >= heights[i]){
                store.pop();
            }

            if(store.isEmpty()){
                leftExtend[i] = i;
            }else{
                leftExtend[i] = i - store.peek() - 1;
            }

            store.push(i);
        }

        store.clear();

        //calculate how much right does this height extend to
        for(int i=n-1;i>=0;i--){
            while(!store.isEmpty() && heights[store.peek()] > heights[i]){
                store.pop();
            }

            if(store.isEmpty()){
                rightExtend[i] = n - i - 1;
            }else{
                rightExtend[i] = store.peek() - i - 1;
            }
            store.push(i);
        }

        int maxArea = Integer.MIN_VALUE;
        for(int i=0;i<heights.length;i++){
             maxArea = Math.max(maxArea, (leftExtend[i]+rightExtend[i]+1)*heights[i]);
        }

        return maxArea;
    }
}
