LeetCode : https://leetcode.com/problems/next-greater-element-i/description/

Solution #1 : Iterative :
Time Complexity : O(nums1.length + nums2.length), Space Complexity : O(nums2.length) :

class Solution {
    public int[] nextGreaterElement(int[] nums1, int[] nums2) {
        HashMap<Integer, Integer> nextGreater = new HashMap<>();
        Stack<Integer> store = new Stack<>();

        for(int i=nums2.length-1;i>=0;i--){
            while(!store.isEmpty() && store.peek() < nums2[i]){
                store.pop();
            }

            if(store.isEmpty()){
                nextGreater.put(nums2[i], -1);
            }else{
                nextGreater.put(nums2[i], store.peek());
            }

            store.push(nums2[i]);
        }

        int[] res = new int[nums1.length];
        for(int i=0;i<nums1.length;i++){
            res[i] = nextGreater.get(nums1[i]);
        }

        return res;
    }
}
