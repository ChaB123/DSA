LeetCode : https://leetcode.com/problems/longest-continuous-subarray-with-absolute-diff-less-than-or-equal-to-limit/description/

Solution #1 :
Time Complexity : O(n), Space Complexity : O(n) :

class Solution {
    public int longestSubarray(int[] nums, int limit) {
        Deque<Integer> dqIncreasing = new ArrayDeque<>();
        Deque<Integer> dqDecreasing = new ArrayDeque<>();
        int ans = 0;
        int left = 0;

        for(int i=0;i<nums.length;i++){
            while(!dqIncreasing.isEmpty() && nums[dqIncreasing.getLast()] >= nums[i]){
                dqIncreasing.pollLast();
            }
            dqIncreasing.addLast(i);

            while(!dqDecreasing.isEmpty() && nums[dqDecreasing.getLast()] <= nums[i]){
                dqDecreasing.pollLast();
            }
            dqDecreasing.addLast(i);

            while(!dqIncreasing.isEmpty() && !dqDecreasing.isEmpty() && nums[dqDecreasing.getFirst()] - nums[dqIncreasing.getFirst()] > limit){
                left += 1;
                while(!dqIncreasing.isEmpty() && dqIncreasing.getFirst() < left){
                    dqIncreasing.pollFirst();
                }

                while(!dqDecreasing.isEmpty() && dqDecreasing.getFirst() < left){
                    dqDecreasing.pollFirst();
                }
            }
            
            ans = Math.max(ans, i-left+1);
        }

        return ans;
    }
}
