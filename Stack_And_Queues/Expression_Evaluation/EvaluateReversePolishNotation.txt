Solution #1 :
Time Complexity : O(n), Space Complexity : O(n) :

class Solution {
    public int evalRPN(String[] tokens) {
        Stack<Integer> store = new Stack<>();

        for(int i=0;i<tokens.length;i++){

            if(tokens[i].length() == 1 && tokens[i].charAt(0) == '+'){
                int rightElement = store.pop();
                int leftElement = store.pop();
                store.push(leftElement+rightElement);
            }else if(tokens[i].length() == 1 && tokens[i].charAt(0) == '-'){
                int rightElement = store.pop();
                int leftElement = store.pop();
                store.push(leftElement-rightElement);
            }else if(tokens[i].length() == 1 && tokens[i].charAt(0) == '*'){
                int rightElement = store.pop();
                int leftElement = store.pop();
                store.push(leftElement*rightElement);
            }else if(tokens[i].length() == 1 && tokens[i].charAt(0) == '/'){
                int rightElement = store.pop();
                int leftElement = store.pop();
                store.push(leftElement/rightElement);
            }else{
                store.push(Integer.parseInt(tokens[i]));
            }
        }

        return store.peek();
    }
}
