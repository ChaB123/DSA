Solution #1 : 
Time Complexity : O(n), Space Complexity : O(n) :

class Solution {
    public int calculate(String s) {

        Stack<String> store = new Stack<>();
        int i = 0;
        boolean doWeNeedToEvaluate = false;
        boolean operatorIncluded = false;

        while(i < s.length()){
            if(s.charAt(i) == ' '){
                i += 1;
                continue;
            }

            if(s.charAt(i) == '+' || s.charAt(i) == '-' || s.charAt(i) == '*' || s.charAt(i) == '/'){
                operatorIncluded = true;

                if(doWeNeedToEvaluate){
                    int rightElement = Integer.parseInt(store.pop());
                    char sign = store.pop().charAt(0);
                    int leftElement = Integer.parseInt(store.pop());
                    int ans = 0;

                    if(sign == '*'){
                        ans = leftElement*rightElement;
                    }else{
                        ans = leftElement/rightElement;
                    }

                    store.push(ans+"");
                }

                store.push(s.charAt(i)+"");

                if(s.charAt(i) == '*' || s.charAt(i) == '/'){
                    doWeNeedToEvaluate = true;
                }else{
                    doWeNeedToEvaluate = false;
                }
            }else{
                if(!store.isEmpty() && (store.peek().charAt(0) != '+' && store.peek().charAt(0) != '-' && store.peek().charAt(0) != '*' && store.peek().charAt(0) != '/')){
                    store.push(store.pop()+s.charAt(i));
                }else{
                    store.push(s.charAt(i)+"");
                }
            }

            i += 1;
        }

        if(doWeNeedToEvaluate){
            int rightElement = Integer.parseInt(store.pop());
            char sign = store.pop().charAt(0);
            int leftElement = Integer.parseInt(store.pop());
            int ans = 0;

            if(sign == '*'){
                ans = leftElement*rightElement;
            }else{
                ans = leftElement/rightElement;
            }

            store.push(ans+"");
        }

        Stack<String> reversedStore = new Stack<>();

        if(operatorIncluded){
            while(!store.isEmpty()){
                reversedStore.push(store.pop());
            }

            while(reversedStore.size() > 1){
                int leftElement = Integer.parseInt(reversedStore.pop());
                char sign = reversedStore.pop().charAt(0);
                int rightElement = Integer.parseInt(reversedStore.pop());
                int ans = 0;

                if(sign == '+'){
                    ans = leftElement+rightElement;
                }else{
                    ans = leftElement-rightElement;
                }

                reversedStore.push(ans+"");
            }

            return Integer.parseInt(reversedStore.peek());
        }

        return Integer.parseInt(store.peek());
    }
}
