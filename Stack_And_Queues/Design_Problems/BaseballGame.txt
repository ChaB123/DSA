Solution #1 : Iterative :
Time Complexity : O(n), Space Complexity : O(n) :

class Solution {
    public int calPoints(String[] operations) {
        Stack<Integer> storeInStack = new Stack<>();

        for(int i=0;i<operations.length;i++){
            char tmp = operations[i].charAt(0);

            if(tmp == '+'){
                int element1 = storeInStack.pop();
                int element2 = storeInStack.peek();
                storeInStack.push(element1);
                storeInStack.push(element1+element2);
            }else if(tmp == 'D'){
                int element1 = storeInStack.peek();
                storeInStack.push(element1*2);
            }else if(tmp == 'C'){
                storeInStack.pop();
            }else{
                storeInStack.push(Integer.parseInt(operations[i]));
            }
        }

        int res = 0;

        while(!storeInStack.isEmpty()){
            //System.out.println("Adding : "+storeInStack.peek());
            res += storeInStack.pop();
        }

        return res;
    }
}
