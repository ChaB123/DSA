Solution #1 : Brute force recursion :
Time Complexity : O(2^(m+n)), Space Complexity : O(m+n) :

class Solution {
    public int longestCommonSubsequence(String text1, String text2) {
        return longestCommonSubsequenceRecursion(text1, text2, 0, 0);
    }
    public int longestCommonSubsequenceRecursion(String str1, String str2, int index1, int index2){
        if(index1 >= str1.length() || index2 >= str2.length()){
            return 0;
        }

        if(str1.charAt(index1) == str2.charAt(index2)){
            return 1+longestCommonSubsequenceRecursion(str1, str2, index1+1, index2+1);
        }

        int increaseStr1 = longestCommonSubsequenceRecursion(str1, str2, index1+1, index2);
        int increaseStr2 = longestCommonSubsequenceRecursion(str1, str2, index1, index2+1);

        return Math.max(increaseStr1, increaseStr2);
    }
}

Solution #2 : Memoization :
Time Complexity : O(m*n), Space Complexity : O(m*n) :

class Solution {
    public int longestCommonSubsequence(String text1, String text2) {
        int[][] arr = new int[text1.length()][text2.length()];

        for(int i=0;i<arr.length;i++){
            for(int j=0;j<arr[0].length;j++){
                arr[i][j] = -1;
            }
        }

        return longestCommonSubsequenceMemoization(text1, text2, 0, 0, arr);
    }
    public int longestCommonSubsequenceMemoization(String str1, String str2, int index1, int index2, int[][] arr){
        if(index1 >= str1.length() || index2 >= str2.length()){
            return 0;
        }

        if(arr[index1][index2] != -1){
            return arr[index1][index2];
        }

        if(str1.charAt(index1) == str2.charAt(index2)){
            arr[index1][index2] = 1+longestCommonSubsequenceMemoization(str1, str2, index1+1, index2+1, arr);
            return arr[index1][index2];
        }

        int increaseStr1 = longestCommonSubsequenceMemoization(str1, str2, index1+1, index2, arr);
        int increaseStr2 = longestCommonSubsequenceMemoization(str1, str2, index1, index2+1, arr);

        arr[index1][index2] = Math.max(increaseStr1, increaseStr2);

        return arr[index1][index2];
    }
}

Solution #3 : DP :
Time Complexity : O(m*n), Space Complexity : O(m*n) :

class Solution {
    public int longestCommonSubsequence(String text1, String text2) {
        return longestCommonSubsequenceDP(text1, text2);
    }
    public int longestCommonSubsequenceDP(String str1, String str2){
        int[][] arr = new int[str1.length()+1][str2.length()+1];

        //initialization
        for(int i=str1.length();i<arr.length;i++){
            for(int j=str2.length();j<arr[0].length;j++){
                arr[i][j] = 0;
            }
        }

        for(int i=str1.length()-1;i>=0;i--){
            for(int j=str2.length()-1;j>=0;j--){
                if(str1.charAt(i) == str2.charAt(j)){
                    arr[i][j] = 1+arr[i+1][j+1];
                }else{
                    arr[i][j] = Math.max(arr[i+1][j], arr[i][j+1]);
                }
            }
        }

        return arr[0][0];
    }
}

Solution #4 : 1D-DP :
Time Complexity : O(m*n), Space Complexity : O(n) :

class Solution {
    public int longestCommonSubsequence(String text1, String text2) {
        return longestCommonSubsequenceDP(text1, text2);
    }
    public int longestCommonSubsequenceDP(String str1, String str2){
        int[] arr = new int[str2.length()+1];

        for(int i=str1.length()-1;i>=0;i--){
            int diagonal = 0;

            for(int j=str2.length()-1;j>=0;j--){
                int oldValue = arr[j];

                if(str1.charAt(i) == str2.charAt(j)){
                    arr[j] = 1+diagonal;
                }else{
                    arr[j] = Math.max(arr[j], arr[j+1]);
                }

                diagonal = oldValue;
            }
        }

        return arr[0];
    }
}
