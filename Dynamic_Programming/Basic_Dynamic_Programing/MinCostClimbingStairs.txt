LeetCode : https://leetcode.com/problems/min-cost-climbing-stairs/description/

Solution #1 : Recursion (below solution will give TLE) :
Time Complexity : O(2^n), Space Complexity : O(n) :

class Solution {
    public int minCostClimbingStairs(int[] cost) {
        int startFrom0 = findMinCost(cost, 0);
        int startFrom1 = findMinCost(cost, 1);

        return Math.min(startFrom0, startFrom1);
    }
    public int findMinCost(int[] cost, int index){
        if(index == cost.length){
            return 0;
        }

        int oneStep = 0;
        int twoStep = 0;

        if(index+1 <= cost.length){
            oneStep = cost[index]+findMinCost(cost, index+1);
        }

        if(index+2 <= cost.length){
            twoStep = cost[index]+findMinCost(cost, index+2);
        }else{
            return oneStep;
        }

        return Math.min(oneStep, twoStep);
    }
}

Solution #2 : Recursion with memoization :
Time Complexity : O(n), Space Complexity : O(n) :

class Solution {
    public int minCostClimbingStairs(int[] cost) {
        int[] arr = new int[cost.length+1];

        for(int i=0;i<arr.length;i++){
            arr[i] = -1;
        }

        int startFrom0 = findMinCost(cost, 0, arr);
        int startFrom1 = findMinCost(cost, 1, arr);

        return Math.min(startFrom0, startFrom1);
    }
    public int findMinCost(int[] cost, int index, int[] arr){
        if(index == cost.length){
            return 0;
        }

        if(arr[index] != -1){
            return arr[index];
        }

        int oneStep = 0;
        int twoStep = 0;

        if(index+1 <= cost.length){
            oneStep = cost[index]+findMinCost(cost, index+1, arr);
        }

        if(index+2 <= cost.length){
            twoStep = cost[index]+findMinCost(cost, index+2, arr);
        }else{
            arr[index] = oneStep;
            return arr[index];
        }

        arr[index] = Math.min(oneStep, twoStep);

        return arr[index];
    }
}

Solution #3 : DP :
Time Compelxity : O(n), Space Complexity : O(n) :

class Solution {
    public int minCostClimbingStairs(int[] cost) {

        int startFrom0 = findMinCost(cost, 0);
        int startFrom1 = findMinCost(cost, 1);

        return Math.min(startFrom0, startFrom1);
    }
    public int findMinCost(int[] cost, int index){
        int n = cost.length;
        int[] arr = new int[n+1];
        

        arr[n] = 0;

        for(int i=n-1;i>=0;i--){
            int oneStep = cost[i]+arr[i+1];

            if(i+2 < n+1){
                int twoStep = cost[i]+arr[i+2];
                arr[i] = Math.min(oneStep, twoStep);
            }else{
                arr[i] = oneStep;
            }    
        }

        return arr[index];
    }
}

class Solution {
    public int minCostClimbingStairs(int[] cost) {
        int n = cost.length;
        int[] arr = new int[n+1];

        arr[n] = 0;

        for(int i=n-1;i>=0;i--){
            int oneStep = cost[i]+arr[i+1];

            if(i+2 < n+1){
                int twoStep = cost[i]+arr[i+2];
                arr[i] = Math.min(oneStep, twoStep);
            }else{
                arr[i] = oneStep;
            }    
        }

        return Math.min(arr[0], arr[1]);
    }
}
