Solution #1 : Recursion :
Time Complexity : O(2^n), Space Complexity : O(n) :

class Solution {
    static boolean isSubsetSum(int arr[], int sum) {
        return findSubset(arr, 0, sum);
    }
    public static boolean findSubset(int[] arr, int index, int sum){
        if(sum == 0 && index <= arr.length){
            return true;
        }
        
        if(index >= arr.length || sum < 0){
            return false;
        }
        
        return findSubset(arr, index+1, sum-arr[index]) || findSubset(arr, index+1, sum);
    }
}

Solution #2 : Transforming "01KnapsackProblem.txt" solution to this question :
Time Complexity : O(n*Sum), Space Complexity : O(n*Sum) :

class Solution {
    static boolean isSubsetSum(int arr[], int sum) {
        boolean[][] knapsack = new boolean[arr.length+1][sum+1];

        for(int i=0;i<knapsack.length;i++){
            for(int j=0;j<knapsack[0].length;j++){
                if(i == 0){
                    knapsack[i][j] = false;
                }

                if(j == 0){
                    knapsack[i][j] = true;
                }
            }
        }

        for(int i=1;i<knapsack.length;i++){
            for(int j=1;j<knapsack[0].length;j++){
                if(arr[i-1] <= j){
                    knapsack[i][j] = knapsack[i-1][j-arr[i-1]] || knapsack[i-1][j];
                }else{
                    knapsack[i][j] = knapsack[i-1][j];
                }
            }
        }

        return knapsack[arr.length][sum];
    }
}
