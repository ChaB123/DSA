Solution #1 : Brute force recursion :
Time Complexity : O(2^n), Space Complexity : O(n) :

class Solution {
    public int lastStoneWeightII(int[] stones) {
        int arraySum = 0;
        for(int i=0;i<stones.length;i++){
            arraySum += stones[i];
        }

        return lastStoneWeightIIRecursion(stones, 0, 0, 0, arraySum);
    }
    public int lastStoneWeightIIRecursion(int[] stones, int index, int currSum, int numOfElement, int sum){

        if(index == stones.length && numOfElement <= (stones.length+1)/2){
            int secondSubSetSum = sum-currSum;

            if(secondSubSetSum > currSum){
                return secondSubSetSum-currSum;
            }

            return currSum-secondSubSetSum;
        }

        if(numOfElement > (stones.length+1)/2){
            return Integer.MAX_VALUE;
        }

        int ifWeInclude = lastStoneWeightIIRecursion(stones, index+1, currSum+stones[index], numOfElement+1, sum);
        int ifWeExclude = lastStoneWeightIIRecursion(stones, index+1, currSum, numOfElement, sum);

        return Math.min(ifWeInclude, ifWeExclude);
    }
}

Solution #2 : Memoization :
Time Complexity : O(n^2 * S) (S = value of array sum), Space Complexity : O(n^2 * S + n) :

class Solution {
    public int lastStoneWeightII(int[] stones) {
        int arraySum = 0;
        for(int i=0;i<stones.length;i++){
            arraySum += stones[i];
        }

        int[][][] memo = new int[stones.length+1][arraySum+1][((stones.length+1)/2)+1];

        for(int i=0;i<memo.length;i++){
            for(int j=0;j<memo[i].length;j++){
                for(int z=0;z<memo[i][j].length;z++){
                    memo[i][j][z] = -1;
                }
            }
        }

        return lastStoneWeightIIRecursion(stones, 0, 0, 0, arraySum, memo);
    }
    public int lastStoneWeightIIRecursion(int[] stones, int index, int currSum, int numOfElement, int sum, int[][][] memo){

        if(index == stones.length && numOfElement <= (stones.length+1)/2){
            int secondSubSetSum = sum-currSum;

            if(secondSubSetSum > currSum){
                memo[index][currSum][numOfElement] = secondSubSetSum-currSum;
            }else{
                memo[index][currSum][numOfElement] = currSum-secondSubSetSum;
            }

            return memo[index][currSum][numOfElement];
        }

        if(numOfElement > (stones.length+1)/2){
            return Integer.MAX_VALUE;
        }

        if(memo[index][currSum][numOfElement] != -1){
            return memo[index][currSum][numOfElement];
        }

        int ifWeInclude = lastStoneWeightIIRecursion(stones, index+1, currSum+stones[index], numOfElement+1, sum, memo);
        int ifWeExclude = lastStoneWeightIIRecursion(stones, index+1, currSum, numOfElement, sum, memo);

        memo[index][currSum][numOfElement] = Math.min(ifWeInclude, ifWeExclude);

        return memo[index][currSum][numOfElement];
    }
}

Solution #3 : Optimised memoization :
Time Complexity : O(n*S), Space Complexity : O(n*S + n) :

class Solution {
    public int lastStoneWeightII(int[] stones) {
        int arraySum = 0;
        for(int i=0;i<stones.length;i++){
            arraySum += stones[i];
        }

        int[][] memo = new int[stones.length+1][arraySum+1];

        for(int i=0;i<memo.length;i++){
            for(int j=0;j<memo[i].length;j++){
                memo[i][j] = -1;
            }
        }

        return lastStoneWeightIIRecursion(stones, 0, 0, arraySum, memo);
    }
    public int lastStoneWeightIIRecursion(int[] stones, int index, int currSum, int sum, int[][] memo){

        if(index == stones.length){
            int secondSubSetSum = sum-currSum;

            if(secondSubSetSum > currSum){
                memo[index][currSum] = secondSubSetSum-currSum;
            }else{
                memo[index][currSum] = currSum-secondSubSetSum;
            }

            return memo[index][currSum];
        }

        if(memo[index][currSum] != -1){
            return memo[index][currSum];
        }

        int ifWeInclude = lastStoneWeightIIRecursion(stones, index+1, currSum+stones[index], sum, memo);
        int ifWeExclude = lastStoneWeightIIRecursion(stones, index+1, currSum, sum, memo);

        memo[index][currSum] = Math.min(ifWeInclude, ifWeExclude);

        return memo[index][currSum];
    }
}

Solution #4 : DP :
Time Complexity : O(n*S), Space Complexity : O(n*S) :

class Solution {
    public int lastStoneWeightII(int[] stones) {
        int arraySum = 0;
        for(int i=0;i<stones.length;i++){
            arraySum += stones[i];
        }

        return lastStoneWeightIIDP(stones, arraySum);
    }
    public int lastStoneWeightIIDP(int[] stones, int currSum){
        int[][] dp = new int[stones.length+1][currSum+1];

        for(int i=dp[0].length-1;i>=0;i--){
            if(currSum-i > i){
                dp[stones.length][i] = currSum-2*i;
            }else{
                dp[stones.length][i] = 2*i-currSum;
            }
        }

        for(int i=dp.length-2;i>=0;i--){
            for(int j=dp[0].length-1;j>=0;j--){
                if(j+stones[i] <= currSum){
                    dp[i][j] = Math.min(dp[i+1][j+stones[i]], dp[i+1][j]);
                }else{
                    dp[i][j] = dp[i+1][j];
                }
            }
        }

        return dp[0][0];
    }
}

Solution #5 : 1D DP :
Time Complexity : O(n*S), Space Complexity : O(S) :

class Solution {
    public int lastStoneWeightII(int[] stones) {
        int arraySum = 0;
        for(int i=0;i<stones.length;i++){
            arraySum += stones[i];
        }

        return lastStoneWeightIIDP(stones, arraySum);
    }
    public int lastStoneWeightIIDP(int[] stones, int currSum){
        int[] dp = new int[currSum+1];

        for(int i=dp.length-1;i>=0;i--){
            if(currSum-i > i){
                dp[i] = currSum-2*i;
            }else{
                dp[i] = 2*i-currSum;
            }
        }

        for(int i=stones.length-1;i>=0;i--){
            for(int j=0;j<dp.length;j++){
                if(j+stones[i] <= currSum){
                    dp[j] = Math.min(dp[j+stones[i]], dp[j]);
                }
            }
        }

        return dp[0];
    }
}
